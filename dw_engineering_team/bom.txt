    def action_create_revision(self):
        for sheet in self:
            new_rev_no = sheet.revision_no + 1

            # 1️⃣ Create revision from CURRENT working copy
            revision = self.env['packing.cost.sheet.revision'].create({
                'sheet_id': sheet.id,
                'revision_no': new_rev_no,
                'revision_name': f"Rev-{new_rev_no}",
            })

            # 2️⃣ Copy CURRENT lines (edited state)
            for line in sheet.line_ids:
                self.env['packing.cost.sheet.revision.line'].create({
                    'revision_id': revision.id,
                    'section': line.section,
                    'description': line.description.id,
                    'length': line.length,
                    'width': line.width,
                    'thickness': line.thickness,
                    'qty': line.qty,
                    'sqft': line.sqft,
                    'material': line.material.id,
                    'rm_rate': line.rm_rate,
                    'amount': line.amount,
                })

            # 3️⃣ Update revision pointer ONLY
            sheet.write({
                'revision_no': new_rev_no,
                'active_revision_id': revision.id,
            })





    def _create_revision_snapshot(self):
        for sheet in self:
            new_rev_no = sheet.revision_no + 1

            revision = self.env['packing.cost.sheet.revision'].create({
                'sheet_id': sheet.id,
                'revision_no': new_rev_no,
                'revision_name': f"Rev-{new_rev_no}",
            })

            for line in sheet.line_ids:
                self.env['packing.cost.sheet.revision.line'].create({
                    'revision_id': revision.id,
                    'section': line.section,
                    'description': line.description.id,
                    'length': line.length,
                    'width': line.width,
                    'thickness': line.thickness,
                    'qty': line.qty,
                    'sqft': line.sqft,
                    'material': line.material.id,
                    'rm_rate': line.rm_rate,
                    'amount': line.amount,
                })

            sheet.revision_no = new_rev_no
            sheet.active_revision_id = revision.id

    @api.depends(
        'line_ids.amount',
        'line_ids.sqft',
        
        'mfg_percent',
        'overhead_percent',
        'profit_percent',
        'freight',
        'packing_charges'
    )
    def _compute_totals(self):
        for rec in self:
            rec.material_total = sum(rec.line_ids.mapped('amount'))
            rec.total_sqft = sum(rec.line_ids.mapped('sqft'))
           

            rec.mfg_amount = rec.material_total * rec.mfg_percent / 100
            rec.overhead_amount = rec.material_total * rec.overhead_percent / 100
            rec.profit_amount = rec.material_total * rec.profit_percent / 100

            rec.grand_total = (
                rec.material_total
                + rec.mfg_amount
                + rec.overhead_amount
                + rec.profit_amount
                + rec.freight
                + rec.packing_charges
            )


    @api.depends('total_cft','total_sqft')
    def _compute_empty_box_weight(self):
        for record in self:
            record.empty_box_weight = (((record.total_cft*17)+(record.total_sqft*0.6))+20)

    ], string="Section")

    length = fields.Float("Length")
    width = fields.Float("Width")
    thickness = fields.Float("Thickness")
    qty = fields.Float("Qty")

    sqft = fields.Float("SQFT/CFT", compute="_compute_area", store=True)
    # cft = fields.Float("CFT", compute="_compute_area", store=True)

    material = fields.Many2one('material.master',string="Material")
    rm_rate = fields.Float("RM Rate")
    amount = fields.Float("Amount", compute="_compute_amount", store=True)

    def _normalize_material_name(self, name):
        if not name:
            return ''
        return re.sub(r'[^a-z0-9]', '', name.lower())


    @api.depends('length', 'width', 'thickness', 'qty', 'material')
    def _compute_area(self):
        for line in self:
            line.sqft = 0.0

            # ---------- HARD SAFETY ----------
            if not line.material:
                continue
            if not line.length or not line.width:
                continue
            if not line.qty or line.qty <= 0:
                continue

            mat_name = self._normalize_material_name(line.material.name)

            # ---------- PINEWOOD ----------
            if 'pinewood' in mat_name:
                if not line.thickness:
                    continue
                line.sqft = (
                    (line.length * line.width * line.thickness)
                    / (305 *305 *305)* (line.qty / 0.95))

            # ---------- PLYWOOD ----------
            elif 'plywood' in mat_name:
                line.sqft = (
                    (line.length / 1000)* (line.width / 1000) * line.qty / 0.95 * 10.764)

            # ---------- HDPE / VCI / MICRON / METER ----------
            elif any(k in mat_name for k in ['hdpe', 'vci', 'micron', 'meter']):
                if not line.thickness:
                    continue
                line.sqft = (
                    (
                        (2 * line.length * line.width)
                        + (2 * line.length * line.thickness)
                        + (2 * line.width * line.thickness)
                    )
                    * line.qty
                )







    @api.depends('sqft', 'rm_rate')
    def _compute_amount(self):
        for line in self:
            line.amount = line.sqft * line.rm_rate

    def _compute_totals(self):
        for rec in self:
            rec.material_total = sum(rec.line_ids.mapped('amount'))
            rec.total_sqft = sum(rec.line_ids.mapped('sqft'))
           

            rec.mfg_amount = rec.material_total * rec.mfg_percent / 100
            rec.overhead_amount = rec.material_total * rec.overhead_percent / 100
            rec.profit_amount = rec.material_total * rec.profit_percent / 100

            rec.grand_total = (
                rec.material_total
                + rec.mfg_amount
                + rec.overhead_amount
                + rec.profit_amount
                + rec.freight
                + rec.packing_charges
            )

        for record in self:
            record.empty_box_weight = (((record.total_cft*17)+(record.total_sqft*0.6))+20)

    # cft = fields.Float("CFT", compute="_compute_area", store=True)

    material = fields.Many2one('material.master',string="Material")
    rm_rate = fields.Float("RM Rate")
    amount = fields.Float("Amount", compute="_compute_amount", store=True)

    def _normalize_material_name(self, name):
        if not name:
            return ''
        return re.sub(r'[^a-z0-9]', '', name.lower())


    @api.depends('length', 'width', 'thickness', 'qty', 'material')
    def _compute_area(self):
        for line in self:
            line.sqft = 0.0

            # ---------- HARD SAFETY ----------
            if not line.material:
                continue
            if not line.length or not line.width:
                continue
            if not line.qty or line.qty <= 0:
                continue

            mat_name = self._normalize_material_name(line.material.name)

            # ---------- PINEWOOD ----------
            if 'pinewood' in mat_name:
                if not line.thickness:
                    continue
                line.sqft = (
                    (line.length * line.width * line.thickness)
                    / (305 *305 *305)* (line.qty / 0.95))

            # ---------- PLYWOOD ----------
            elif 'plywood' in mat_name:
                line.sqft = (
                    (line.length / 1000)* (line.width / 1000) * line.qty / 0.95 * 10.764)

            # ---------- HDPE / VCI / MICRON / METER ----------
            elif any(k in mat_name for k in ['hdpe', 'vci', 'micron', 'meter']):
                if not line.thickness:
                    continue
                line.sqft = (
                    (
                        (2 * line.length * line.width)
                        + (2 * line.length * line.thickness)
                        + (2 * line.width * line.thickness)
                    )
                    * line.qty
                )







    @api.depends('sqft', 'rm_rate')
    def _compute_amount(self):
        for line in self:
            line.amount = line.sqft * line.rm_rate
